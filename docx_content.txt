2.2 Cài Docker Engine
Docker Engine không có sẵn trong repo Ubuntu. Cần thêm repo chính thức của Docker rồi mới cài.

# Buoc 1: Them GPG key cua Docker (de xac thuc package)
curl -fsSL https://download.docker.com/linux/ubuntu/gpg | \
  sudo gpg --dearmor -o /etc/apt/keyrings/docker.gpg
 
# Buoc 2: Them Docker repository vao sources list
# - $(dpkg --print-architecture): lay CPU arch (amd64, arm64)
# - $(lsb_release -cs): lay Ubuntu codename (jammy, focal, noble)
echo "deb [arch=$(dpkg --print-architecture) \
  signed-by=/etc/apt/keyrings/docker.gpg] \
  https://download.docker.com/linux/ubuntu \
  $(lsb_release -cs) stable" | \
  sudo tee /etc/apt/sources.list.d/docker.list > /dev/null
 
# Buoc 3: Cai Docker
sudo apt-get update
sudo apt-get install -y \
  docker-ce \           # Docker Engine (daemon)
  docker-ce-cli \        # CLI: lenh "docker"
  containerd.io \        # Container runtime
  docker-compose-plugin  # Plugin: lenh "docker compose" (v2)
 
# Buoc 4: Bat tu dong start khi reboot
sudo systemctl enable docker
sudo systemctl start docker
 
# Kiem tra:
docker --version         # Docker version 26.x.x
docker compose version   # Docker Compose version v2.x.x

PHẦN III: PHÂN VÙNG Ổ DATA
Chỉ nếu có ổ thứ 2 riêng — bỏ qua nếu chỉ 1 ổ

ℹ️ Nếu chỉ có 1 ổ đĩa: bỏ qua Phần III. Docker tự tạo volume trên ổ OS.
Nếu có ổ thứ 2 riêng (không có mountpoint): làm Phần III để tách data ra ổ riêng, an toàn hơn khi OS hỏng.

3.1 Xác định ổ data

# Xem tat ca o dia:
lsblk -o NAME,SIZE,TYPE,MOUNTPOINT,FSTYPE
 
# Giai thich output - vi du:
NAME           SIZE TYPE MOUNTPOINT  FSTYPE
sda            60G  disk             <- O TRONG (khong co mountpoint) = o data
sdb           100G  disk
|-sdb1          1G  part /boot/efi   vfat   <- O OS
|-sdb2          2G  part /boot       ext4   <- O OS
|-sdb3         97G  part             LVM2_member
  |-ubuntu--lv 97G  lvm  /           ext4   <- O OS (co /)
 
# O data = o KHONG CO mountpoint VA KHONG CO FSTYPE
# Trong vi du: sda la o data


⚠️ XÁC ĐỊNH ĐÚNG ổ data trước khi chạy lệnh format!
Format nhầm ổ có / (OS) sẽ MẤT TOÀN BỘ HỆ THỐNG không khôi phục được.
Khi không chắc: hỏi lại, không đoán.

3.2 Phân vùng, format, mount

# === THAY sda BANG TEN O THUC TE CUA BAN ===
 
# 1. Tao bang phan vung GPT va 1 partition chiem toan bo o:
# parted: cong cu quan ly partition
# --script: chay khong can nhap them
# mklabel gpt: tao bang phan vung GPT (hien dai, ho tro o lon)
# mkpart primary ext4 0% 100%: 1 partition tu dau den cuoi
sudo parted /dev/sda --script mklabel gpt mkpart primary ext4 0% 100%
 
# 2. Format partition bang ext4:
# -F: force, khong hoi xac nhan
sudo mkfs.ext4 -F /dev/sda1
 
# 3. Lay UUID de dung trong fstab:
sudo blkid -s UUID -o value /dev/sda1
# Ket qua vi du: abc12345-1234-1234-1234-abcdef123456
# Luu gia tri nay lai de dung o buoc 5
 
# 4. Tao thu muc mount:
sudo mkdir -p /data/postgres /data/redis
 
# 5. Them vao /etc/fstab (thay UUID bang gia tri thuc te):
# fstab: file cau hinh tu dong mount khi boot
# Cac option: defaults=rw,exec,auto,nouser,async
# noatime: khong cap nhat access time (tang hieu suat)
# 0 2: khong backup, kiem tra sau /
echo "UUID=abc12345-1234-1234-1234-abcdef123456 /data ext4 defaults,noatime 0 2" | \
  sudo tee -a /etc/fstab
 
# 6. Mount ngay khong can reboot:
sudo mount -a
 
# 7. Phan quyen cho PostgreSQL (chay voi UID 999 trong container):
sudo chown -R 999:999 /data/postgres /data/redis
 
# Kiem tra thanh cong:
df -h | grep /data
# Phai thay: /dev/sda1    98G   24K   93G   1%   /data

📄  app/utils/auth.py   —   JWT token, bcrypt hash, dependency injection cho FastAPI
get_current_user và require_admin được inject vào mọi endpoint cần xác thực
from jose import JWTError, jwt
from passlib.context import CryptContext
from fastapi import Depends, HTTPException
from fastapi.security import OAuth2PasswordBearer
from sqlalchemy.orm import Session
from ..database import get_db
from ..models.user import User
from ..config import settings
from datetime import datetime, timedelta, timezone
pwd_ctx = CryptContext(schemes=["bcrypt"], deprecated="auto")
oauth2 = OAuth2PasswordBearer(tokenUrl="/api/auth/login")
def verify_password(p, h): return pwd_ctx.verify(p, h)
def hash_password(p): return pwd_ctx.hash(p)
def create_access_token(data: dict) -> str:
    exp = datetime.now(timezone.utc) + timedelta(minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES)
    return jwt.encode({**data, "exp": exp}, settings.SECRET_KEY, algorithm=settings.ALGORITHM)
def get_current_user(token: str = Depends(oauth2), db: Session = Depends(get_db)):
    try:
        payload = jwt.decode(token, settings.SECRET_KEY, algorithms=[settings.ALGORITHM])
        user = db.query(User).filter(User.id == int(payload["sub"])).first()
        if not user or not user.is_active: raise HTTPException(401)
        return user
    except JWTError: raise HTTPException(401, "Invalid token")
def require_admin(user=Depends(get_current_user)):
    if user.role != "admin": raise HTTPException(403, "Admin required")
    return user

📄  app/routers/auth.py   —   POST /api/auth/login — xác thực và trả JWT
Nhận form-data (username + password), trả access_token + role + full_name
from fastapi import APIRouter, Depends, HTTPException
from fastapi.security import OAuth2PasswordRequestForm
from sqlalchemy.orm import Session
from ..database import get_db
from ..models.user import User
from ..utils.auth import verify_password, create_access_token
router = APIRouter()
@router.post("/login")
def login(form: OAuth2PasswordRequestForm = Depends(), db: Session = Depends(get_db)):
    user = db.query(User).filter(User.email == form.username).first()
    if not user or not verify_password(form.password, user.hashed_pw):
        raise HTTPException(401, "Invalid email or password")
    if not user.is_active: raise HTTPException(403, "Account disabled")
    token = create_access_token({"sub": str(user.id), "role": user.role})
    return {"access_token": token, "token_type": "bearer", "role": user.role, "full_name": user.full_name}


📄  app/routers/users.py   —   CRUD users — chỉ Admin dùng được

from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.orm import Session
from ..database import get_db
from ..models.user import User
from ..utils.auth import require_admin, hash_password, get_current_user
router = APIRouter()
@router.get("/")
def list_users(db: Session = Depends(get_db), _=Depends(require_admin)):
    return [{"id":u.id,"email":u.email,"full_name":u.full_name,"role":u.role,"is_active":u.is_active} for u in db.query(User).all()]
@router.post("/")
def create_user(body: dict, db: Session = Depends(get_db), _=Depends(require_admin)):
    if db.query(User).filter(User.email == body["email"]).first(): raise HTTPException(400, "Email exists")
    u = User(email=body["email"], full_name=body.get("full_name",""), hashed_pw=hash_password(body["password"]), role=body.get("role","group_a"))
    db.add(u); db.commit(); db.refresh(u)
    return {"id":u.id,"email":u.email,"role":u.role}
@router.put("/{uid}")
def update_user(uid: int, body: dict, db: Session = Depends(get_db), _=Depends(require_admin)):
    u = db.query(User).filter(User.id == uid).first()
    if not u: raise HTTPException(404)
    for k, v in body.items():
        if k == "password": setattr(u, "hashed_pw", hash_password(v))
        elif hasattr(u, k) and k not in ["id","hashed_pw"]: setattr(u, k, v)
    db.commit(); return {"message": "Updated"}
@router.delete("/{uid}")
def delete_user(uid: int, db: Session = Depends(get_db), _=Depends(require_admin)):
    u = db.query(User).filter(User.id == uid).first()

def get_service():
    creds = service_account.Credentials.from_service_account_file(
        settings.GOOGLE_APPLICATION_CREDENTIALS,
        scopes=["https://www.googleapis.com/auth/spreadsheets.readonly"])
    return build("sheets", "v4", credentials=creds, cache_discovery=False)
 
def read_tabs(spreadsheet_id: str, required_cols: list, tab_names: list = None) -> list:
    svc = get_service()
    meta = svc.spreadsheets().get(spreadsheetId=spreadsheet_id).execute()
    req_set = set(c.strip().upper() for c in required_cols)
    results = []
    for sheet in meta.get("sheets", []):
        tab = sheet["properties"]["title"]
        if tab_names and tab not in tab_names: continue
        data = svc.spreadsheets().values().get(spreadsheetId=spreadsheet_id, range=tab).execute()
        rows = data.get("values", [])
        if not rows: continue
        headers_up = [h.strip().upper() for h in rows[0]]
        if not req_set.issubset(set(headers_up)): continue
        headers = [h.strip() for h in rows[0]]
 
        # Tim cot Task ID (cot dau hoac cot co "ID" trong ten)
        task_id_col = None
        for h in headers:
            if h.strip().upper() in ["TASK ID","ID","TASK_ID","#","NO","STT"]:
                task_id_col = h.strip(); break
        if task_id_col is None and headers:
            task_id_col = headers[0]
 
        tab_rows = []
        for i, row in enumerate(rows[1:], start=2):
            padded = row + [""] * (len(headers) - len(row))
            d = dict(zip(headers, padded)); d["_row"] = i
            if not any(str(v).strip() for k,v in d.items() if k != "_row"): continue
            # Chi check dong co Task ID la so thuc su (bo qua header mau, dong tieu de)
            task_id_val = str(d.get(task_id_col, "")).strip() if task_id_col else ""
            if not task_id_val: continue
            if not any(c.isdigit() for c in task_id_val): continue
            tab_rows.append(d)
        if tab_rows:
            results.append({"tab_name": tab, "rows": tab_rows})
    return results

📄  app/worker/policy_engine.py   —   Kiểm tra rule cứng — không tốn AI token
Chạy trước AI. Rule: manday_max, manday_min, min_words, required_fields
def check_row(row: dict, policy: dict, required_cols: list) -> list:
    violations = []
    def add(code, msg): violations.append({"code":code,"message":msg})
    col_map = {}
    for rc in required_cols:
        for k in row.keys():
            if k.strip().upper() == rc.strip().upper(): col_map[rc.strip().upper()] = k; break
    for rule in policy.get("rules", []):
        field = rule.get("field","").strip().upper()
        col_name = col_map.get(field, field)
        cell_val = str(row.get(col_name,"")).strip()
        rule_val = str(rule.get("value","")).strip()
        if rule_val and cell_val.upper() != rule_val.upper(): continue
        if rule.get("manday_max") is not None:
            md_col = col_map.get("MANDAY (EST)","MANDAY (EST)")
            try:
                mv = float(str(row.get(md_col,0) or 0).replace(",","."))
                if mv > rule["manday_max"]: add("MANDAY_TOO_HIGH",f"Manday={mv} exceeds max {rule['manday_max']}")
            except: add("MANDAY_NOT_NUMBER","Manday must be a number")
        if rule.get("manday_min") is not None:
            md_col = col_map.get("MANDAY (EST)","MANDAY (EST)")
            try:
                mv = float(str(row.get(md_col,0) or 0).replace(",","."))
                if mv < rule["manday_min"]: add("MANDAY_TOO_LOW",f"Manday={mv} below min {rule['manday_min']}")
            except: pass
        if rule.get("min_words") is not None:
            task_col = col_map.get("DETAIL TASK","DETAIL TASK")
            tv = str(row.get(task_col,"")).strip()
            if len(tv.split()) < rule["min_words"]: add("DESC_TOO_SHORT",f"Need at least {rule['min_words']} words")
        for rf in rule.get("required_fields",[]):
            rc2 = col_map.get(rf.strip().upper(), rf)
            if not str(row.get(rc2,"")).strip(): add("MISSING_FIELD",f"Required: {rf}")
    return violations



Cách rule hoạt động
Mỗi rule có: field (cột), value (giá trị trigger), và các điều kiện kiểm tra. Rule chỉ chạy khi các cột field=value. Nếu value để trống → áp dụng cho tất cả dòng.

from ..config import settings

DEFAULT_PROMPT = """You are a Project Manager and Task Quality Evaluator for an Infrastructure/SI team.
 
IMPORTANT: Only evaluate tasks that have a Task ID (first column is not empty).
If Task ID is empty, return immediately: {"verdict": "SKIP", "reason": "No Task ID", "suggestion": ""}
 
TEAM CONTEXT:
This is an Infrastructure/SI team that sometimes handles tasks outside pure infrastructure scope (UI changes, app configuration, integrations, documentation). When evaluating such tasks:
- Consider that infra engineers doing non-core tasks face higher complexity than specialists would
- UI/UX issues CAN affect system availability - a broken interface means users cannot access the system
- Do NOT automatically downgrade priority just because a task is "not pure infra"
- Evaluate actual business and system impact, not just the technical category
 
CLASSIFICATION POLICY:
 
CRITICAL (20 KPI/manday) - Emergency in progress + significant impact
  * Production down, widespread outage, core transactions failing, service completely unavailable
  * Security incident: credential leak, ransomware, actively exploited CVE, compromised account
  * Hard deadline: go-live, cutover, major business event - delay causes direct financial/operational loss
  * UI/system issue that blocks ALL users from accessing a critical business function
  * Required evidence: incident/alert ID + affected services + user count + SLA or deadline
  * Typical manday: 0.5-3 days (emergencies are short and focused)
 
HIGH (12 KPI/manday) - Structural change OR significant incident
  * P2/P3 incident: intermittent service, latency spike, partial degradation, not full outage
  * Architecture work: HA setup, network segmentation, auth redesign, pipeline overhaul
  * Critical component migration, Terraform/Ansible changes affecting production, RCA
  * UI/app changes that affect core user workflows, business processes, or system integrations
  * Required evidence: ticket/CR + scope + expected impact + rollback plan
  * Typical manday: 1-5 days; more than 7 days requires clear justification
 
NORMAL (6 KPI/manday) - Routine, SOP-based, low risk
  * Following existing checklist or playbook, small config change, easy to rollback
  * No production impact or very limited/controlled impact
  * User/role management, backup testing, DNS updates, monitoring dashboards, documentation
  * Minor UI tweaks that do not affect core workflows or business functions
  * Typical manday: 0.5-3 days; more than 3 days requires justification
 
EVALUATION RULES:
1. Read the task description carefully - imagine what this task actually involves and how long it realistically takes
2. Consider the perspective of an infra engineer doing this task
3. Verify that Priority matches the actual work and its real impact
4. Verify that Manday is realistic - slightly higher is acceptable for out-of-scope tasks
5. If description is vague (fewer than 5 meaningful words): mark as REVIEW
6. A task cannot be Critical just because someone labeled it so - it requires evidence
7. If Priority seems too LOW for the described work, flag as FAIL and suggest upgrading
 
RESPOND IN VALID JSON ONLY - no markdown, no text outside JSON:
{
  "verdict": "PASS or FAIL or REVIEW or SKIP",
  "reason": "1-2 sentences in Vietnamese: what the task does and why this verdict",
  "suggestion": "Specific advice in Vietnamese: for FAIL state exactly what to change and why; for REVIEW state what is missing; for PASS confirm why it meets standards; for SKIP leave empty"
}"""
 
def get_ai_cfg():
    db = SessionLocal()
    try:
        cfg = get_setting("ai_config", db) or {}
        return {
            "api_key":       cfg.get("api_key")       or settings.AI_API_KEY,
            "base_url":      cfg.get("base_url")      or settings.AI_BASE_URL,
            "model":         cfg.get("model")         or settings.AI_MODEL,
            "system_prompt": cfg.get("system_prompt") or DEFAULT_PROMPT
        }
    finally: db.close()
 
def evaluate_row(row: dict, required_cols: list) -> dict:
    cfg = get_ai_cfg()
    row_str = " | ".join(f"{k}: {v}" for k,v in row.items() if k != "_row" and v)
    try:
        r = requests.post(
            f"{cfg['base_url']}/chat/completions",
            headers={"Authorization":f"Bearer {cfg['api_key']}","Content-Type":"application/json"},
            json={"model":cfg["model"],"messages":[
                {"role":"system","content":cfg["system_prompt"]},
                {"role":"user","content":f"Evaluate this task:\n{row_str}\n\nReturn JSON only."}
            ],"temperature":0.1},
            timeout=settings.AI_TIMEOUT_SECONDS)
        content = r.json()["choices"][0]["message"]["content"].strip()
        content = re.sub(r"```json\s*|\s*```","",content).strip()
        parsed = json.loads(content)
        return {"verdict":parsed.get("verdict","REVIEW"),"reason":parsed.get("reason",""),"suggestion":parsed.get("suggestion","")}
    except Exception as e:
        return {"verdict":"REVIEW","reason":f"AI error: {str(e)[:100]}","suggestion":""}



Tại sao không dùng OpenAI SDK?
OpenAI SDK phiên bản mới (≥40.0) có tham số proxies xung đột với httpx trong container. Lỗi: TypeError: Client.__init__() got an unexpected keyword argument proxies.
Giải pháp: dùng requests.post() trực tiếp, tương thích mọi phiên bản.

Tại sao không dùng response_format: json_object?
ShopAIKey không hỗ trợ response_format với tất cả model. Một số model bị treo hoàn toàn khi gửi tham số này.
Giải pháp: yêu cầu AI trong prompt trả JSON thuần, dùng regex strip markdown, sau đó json.loads().

import uuid, json, logging, redis as redis_lib, os
 
logger = logging.getLogger(__name__)
 
def get_redis():
    return redis_lib.from_url(os.environ.get("REDIS_URL","redis://localhost:6379/0"), decode_responses=True)
 
def push_log(sheet_id, msg, level="info"):
    try:
        r = get_redis()
        entry = json.dumps({"ts":datetime.now().strftime("%H:%M:%S"),"msg":msg,"level":level})
        r.rpush(f"check_log:{sheet_id}", entry)
        r.expire(f"check_log:{sheet_id}", 3600)
        r.set(f"check_status:{sheet_id}", "running", ex=3600)
    except: pass
 
def set_status(sheet_id, status):
    try: get_redis().set(f"check_status:{sheet_id}", status, ex=3600)
    except: pass
 
@celery_app.task(name="app.worker.tasks.check_sheet", bind=True, max_retries=2)
def check_sheet(self, spreadsheet_id: str, sheet_id: int, run_id: str):
    db = SessionLocal()
    try:
        try: r=get_redis(); r.delete(f"check_log:{sheet_id}"); r.set(f"check_status:{sheet_id}","running",ex=3600)
        except: pass
 
        push_log(sheet_id, "Starting check...")
        col_cfg       = get_setting("column_config", db) or {}
        required_cols = col_cfg.get("cols", ["DETAIL TASK","PRIORITY","MANDAY (EST)","STATUS","ASSIGNED"])
        tab_names     = col_cfg.get("tab_names", [])
        policy        = get_setting("policy", db) or {"rules":[]}
 
        deleted = db.query(Violation).filter(Violation.sheet_id == sheet_id).delete()
        db.commit()
        push_log(sheet_id, f"Cleared {deleted} previous violations")
 
        push_log(sheet_id, "Reading Google Sheet...")
        tabs = read_tabs(spreadsheet_id, required_cols, tab_names or None)
        total = sum(len(t["rows"]) for t in tabs)
        push_log(sheet_id, f"Found {len(tabs)} tabs: {', '.join(t['tab_name'] for t in tabs)}")
        push_log(sheet_id, f"Total rows with Task ID: {total}")
 
        processed = 0; saved = 0
        for tab in tabs:
            push_log(sheet_id, f"── Tab: {tab['tab_name']} ({len(tab['rows'])} rows)")
            for row in tab["rows"]:
                processed += 1
                rule_viols = check_row(row, policy, required_cols)
                ai_result  = evaluate_row(row, required_cols)
 
                # Bo qua PASS va SKIP - khong luu vao DB
                if ai_result["verdict"] in ("PASS","SKIP") and not rule_viols:
                    continue
 
                codes = rule_viols if rule_viols else [{"code":"AI_REVIEW","message":""}]
                for v in codes:
                    db.add(Violation(
                        sheet_id=sheet_id, tab_name=tab["tab_name"],
                        row_number=row["_row"],
                        row_data=json.dumps(row, ensure_ascii=False),
                        violation_code=v["code"], violation_msg=v["message"],
                        ai_verdict=ai_result["verdict"],
                        ai_reason=ai_result.get("reason",""),
                        ai_suggestion=ai_result.get("suggestion",""),
                        check_run_id=run_id))
                    saved += 1
                    desc = str(row.get("DETAIL TASK",""))[:35]
                    icon = "X" if ai_result["verdict"]=="FAIL" else "!"
                    push_log(sheet_id, f"[{icon}] Row {row['_row']}: [{ai_result['verdict']}] {desc}",
                             "warn" if ai_result["verdict"]=="FAIL" else "info")
 
            db.commit()
            push_log(sheet_id, f"Tab {tab['tab_name']} done - {saved} violations so far", "success")
 
        s = db.query(Sheet).filter(Sheet.id == sheet_id).first()
        if s: s.last_checked = datetime.now(timezone.utc); db.commit()
 
        push_log(sheet_id, f"DONE: {processed} rows checked, {saved} violations found", "success")
        set_status(sheet_id, "done")
        logger.info(f"[DONE] Sheet {sheet_id}: {processed} rows, {saved} violations")
 
    except Exception as exc:
        push_log(sheet_id, f"ERROR: {str(exc)[:150]}", "error")
        set_status(sheet_id, "error")
        db.rollback()
        raise self.retry(exc=exc, countdown=30)
    finally: db.close()
 
@celery_app.task(name="app.worker.tasks.check_all_sheets")
def check_all_sheets():
    db = SessionLocal()
    try:
        cfg = get_setting("ai_config", db) or {}
        interval_h = cfg.get("check_interval_hours", 1)
        from datetime import timedelta
        cutoff = datetime.now(timezone.utc) - timedelta(hours=interval_h)
        for s in db.query(Sheet).filter(Sheet.is_active==True).all():
            lc = s.last_checked
            if lc is None or (lc.tzinfo is None and lc < cutoff.replace(tzinfo=None)) or \
               (lc.tzinfo is not None and lc < cutoff):
                check_sheet.delay(s.spreadsheet_id, s.id, str(uuid.uuid4()))
    finally: db.close()

PHẦN XII: NGINX, ĐỔI DOMAIN, CHECKLIST

12.1 Vấn đề khi dùng domain public
Nếu frontend build với BASE = http://10.120.40.27:8000/api thì user ngoài mạng không đưa request đến được backend (IP nội bộ không accessible từ internet).
Giải pháp: Nginx đóng vai trò reverse proxy, forward cả / lẫn /api/ vào VM nội bộ.
📌 Thứ tự đúng: (1) Test IP nội bộ OK → (2) Thêm /api/ vào Nginx → (3) Rebuild frontend với domain
Không cần domain để hệ thống chạy nội bộ.

12.2 Thêm location /api/ vào Nginx (trên Nginx server)

# Them truoc "location /" hien tai:
server {
    server_name check.securityzone.vn;
 
    # === THEM DOAN NAY ===
    location /api/ {
        proxy_pass http://10.120.40.27:8000/api/;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto https;
        proxy_read_timeout 120s;     # AI goi lau, can cao hon 60s mac dinh
        proxy_connect_timeout 10s;
    }
    # === KET THUC THEM ===
 
    location / {
        proxy_pass http://10.120.40.27:3000;
        # ... giu nguyen ...
    }
 
    # ... phan SSL giu nguyen ...
}
 
# Test va reload Nginx (tren Nginx server):
nginx -t && systemctl reload nginx

Lệnh debug nhanh

# Xem log worker theo thoi gian thuc:
docker logs task_worker -f --tail=30
 
# Xem loi backend:
docker logs task_backend --tail=50 2>&1 | grep -E "ERROR|Error|Exception|Traceback"
 
# Flush Redis (xoa job cu bi loi):
docker exec task_redis redis-cli FLUSHALL
 
# Restart 1 container:
docker restart task_backend
 
# Restart tat ca:
cd /opt/task-portal/app && docker compose restart
 
# Xem violations trong DB:
docker exec task_postgres psql -U task_user -d task_portal \
  -c "SELECT ai_verdict, COUNT(*) FROM violations GROUP BY ai_verdict;"
 
# Kiem tra sheet trong DB:
docker exec task_postgres psql -U task_user -d task_portal \
  -c "SELECT id, name, last_checked FROM sheets;"
 
# Xem setting hien tai:
docker exec task_postgres psql -U task_user -d task_portal \
  -c "SELECT key, LEFT(value,100) FROM settings;"
 
# Test AI API truc tiep:
KEY=$(grep AI_API_KEY /opt/task-portal/app/.env | cut -d= -f2)
curl -s https://api.shopaikey.com/v1/models -H "Authorization: Bearer $KEY" | \
  python3 -c "import sys,json; [print(m['id']) for m in json.load(sys.stdin)['data'][:5]]"
